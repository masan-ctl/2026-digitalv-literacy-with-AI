// 1. 서버 데이터 불러오기 함수 수정 async function fetchServerData(showLoading = false) { if (!GOOGLE_SCRIPT_URL) { renderCalendar(); return; } if (showLoading) { document.getElementById("loadingOverlay").style.display = "flex"; } try { // redirect: "follow" 추가하여 구글 앱스 스크립트 응답 정상 수신 const res = await fetch(GOOGLE_SCRIPT_URL, { redirect: "follow" }); const data = await res.json(); applications = Array.isArray(data) ? data : []; console.log("불러온 신청 내역:", applications); // 브라우저 콘솔에서 확인 가능 } catch (err) { console.error("데이터 로딩 실패:", err); } finally { if (showLoading) { document.getElementById("loadingOverlay").style.display = "none"; } renderCalendar(); } } // 2. 슬롯 계산 함수 비교 로직 강화 function getSlotStats(dateStr, timeStr) { // 공백 제거 및 텍스트 일치 여부 정확히 비교 const list = applications.filter(a => { const aDate = String(a.date || '').trim(); const aTime = String(a.time || '').replace(/\s+/g, ''); const targetTime = String(timeStr || '').replace(/\s+/g, ''); return aDate === dateStr && aTime === targetTime; }); const room1Count = list.filter(a => String(a.room).trim() === "강의실1").reduce((sum, a) => sum + Number(a.count || 0), 0); const room2Count = list.filter(a => String(a.room).trim() === "강의실2").reduce((sum, a) => sum + Number(a.count || 0), 0); const totalCount = room1Count + room2Count; return { room1Count, room2Count, totalCount, list }; } // 3. 신청서 제출 함수 수정 async function handleApplySubmit(e) { e.preventDefault(); const date = document.getElementById("appDate").value; const time = document.getElementById("appTime").value; const room = document.getElementById("appRoom").value; const count = parseInt(document.getElementById("appCount").value); const dept = document.getElementById("appDept").value.trim(); const className = document.getElementById("appClass").value.trim(); const priority2 = document.getElementById("appPriority2").value.trim(); const priority3 = document.getElementById("appPriority3").value.trim(); const note = document.getElementById("appNote").value.trim(); const name = document.getElementById("appName").value.trim(); const pw = document.getElementById("appPw").value.trim(); if (HOLIDAYS[date] || date < config.startDate || date > config.endDate) { alert("신청할 수 없는 일자입니다."); return; } const formData = { id: Date.now().toString(), date, time, room, count, dept, className, priority2, priority3, note, name, pw }; document.getElementById("loadingOverlay").style.display = "flex"; try { // text/plain 타입으로 전송하여 CORS Preflight 에러 방지 await fetch(GOOGLE_SCRIPT_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(formData), redirect: "follow" }); alert("신청이 접수되었습니다!"); resetForm(); setTimeout(() => fetchServerData(false), 1500); } catch (err) { alert("제출 중 오류가 발생했습니다."); console.error(err); } finally { document.getElementById("loadingOverlay").style.display = "none"; } }
